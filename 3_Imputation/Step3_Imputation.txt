##### GWAS Array QC #####
# This tutorial is designed to demonstrate how to perform imputation on QCed SNP array data.

# In order to use this tutorial, you will need to determine a name for your project (ProjectName).

# The inputs and outputs for each step are listed.

#################################
##### Step 1: Prepare Data ######
#################################

# Several tasks must be completed before submitting data to the imputation server.
# Data must be separated by chromosome and must be in vcf format.

# The Pre-imputation_MJ.bash script does the following:
#	remove exclude list variants from dataset
#	perform several filters from gnomAD
#	split dataset into chromosomes
#	convert plink files to vcfs
#	name variants with chr:pos:ref:alt format
#	set heterozygous haplotypes to missing on the X chromosome

# Inputs:
# 	The output plink files from the previous QC step.
#	Project Name
#	A combined SNP exclusion list - collect all SNPs from exclude lists into a single file. This file is one column with variant IDs.
# Outputs:
#	vcf files for each chromosome

export NAMEBASE="YourProjectName"

# Remove variants from pseudoGWAS blacklist. Note the /path/to/previousdata is the path to the output of the previous tutorial without a file extension
plink2 --bfile /path/to/previousdata --exclude /path/to/blacklist.txt --make-bed --out ${NAMEBASE}_excluded

# Perform for all chromosomes individually
for CHR in {1..22} X; do

# Exclude variants from ranges defined by gnomAD
plink2 --bfile ${NAMEBASE}_excluded \
	--chr $CHR \
	--exclude range \
	gnomAD_QC/LCR_filter_gnomAD_v3/filtered_position_hg38/${CHR}_hg38_LCR_gnomAD_v3.txt \
	gnomAD_QC/MultiAllelic_and_InsDel_filter_gnomAD_v3/filtered_position_hg38/${CHR}_hg38_MultiAllelic_and_InsDel_gnomAD_v3.txt \
	gnomAD_QC/PrimerPoly_5perc_filter_gnomAD_v3/filtered_position_hg38/${CHR}_hg38_PrimerPoly_5perc_gnomAD_v3.txt \
	gnomAD_QC/SV_filter_gnomAD_v2/filtered_position_hg38/hg38_SV_gnomAD_v2.txt \
	--output-chr chrM \
	--set-hh-missing \
	--recode vcf bgz id-paste=iid \
	--out ${NAMEBASE}_excluded.chr${CHR}

bcftools annotate --set-id '%CHROM\:%POS\:%REF\:%ALT' -Oz -o ${NAMEBASE}_excluded.chr${CHR}.SNPID.vcf.gz ${NAMEBASE}_excluded.chr${CHR}.vcf.gz

rm ${OUTDIR}/${NAMEBASE}_excluded.chr${CHR}.vcf.gz

done

# The outcome is a set of imputation server-ready vcf files.


##############################
##### Step 2: Imputation #####
##############################

# Use an imputation server to impute data from our QCed files
# TOPMed's Imputation Server is free to use at https://imputation.biodatacatalyst.nhlbi.nih.gov/
# Instructions refer to build 2.0.0-beta3

# Go to the imputation server site and log in or create an account.
# Once logged in, navigate to Run -> Genotype Imputation (Minimac4)

# Select "TOPMed r3" for "Reference Panel"
# Click select files and use CTRL to select all of the vcf files we created, including chrX
# Select "GRCh38/hg38" for "Array Build"
# Select "0.3" for "rsq Filter"
# Select "Eagle v2.4 (phased output)" for "Phasing Engine"
# Select "vs. TOPMed Panel" for "Allele Frequency Check"
# Select "Quality Control & Imputation" for "Mode"
# Check the two boxes that start with "I will"
# Click "Submit Job"

# Your data will be QCed and imputed by the server. Once done, you will receive an email with a password and a link to your data.


######################################
##### Step 3: Post-Imputation QC #####
######################################

# Data from the imputation server requires download and processing.
# Follow the link in the imputation completion email.
# Download the available files.

# The Post-imputation_MJ.bash script does the following:
#	unzip output files
#	filter for Rsq > 0.7
#	set variant IDs to chr:pos:ref:alt format
#	gathers output vcfs into a single vcf
#	convert to plink files

# Inputs:
#	Project Name
#	Password provided in email. This should be in single quotes and any special characters need to be escaped with a \
# Outputs:
#	A set of imputed plink files

# Download the data using either the curl code or wget code provided by the imputation server.

# Decompress imputation output
parallel -j 23 "unzip -P 'yourpassword' chr_{1}.zip" ::: {1..22} X

# Filter for Rsq > 0.7
parallel -j 23 "bcftools view -i 'INFO/R2 > 0.7' -Oz -o chr{1}_r20.7.dose.vcf.gz chr{1}.dose.vcf.gz" ::: {1..22} X

# Add chr:pos:ref:alt SNPIDs
parallel -j 23 "bcftools annotate --set-id '%CHROM:%POS:%REF:%ALT' -Oz -o chr{1}_r20.7_annSNPID.vcf.gz chr{1}_r20.7.dose.vcf.gz" ::: {1..22} X

parallel -j 23 "bcftools head chr{1}_r20.7_annSNPID.vcf.gz | grep -v contig > chr{1}_r20.7_annSNPID.hdr" ::: {1..22} X

parallel -j 23 "bcftools reheader --header chr{1}_r20.7_annSNPID.hdr -o chr{1}_r20.7_annSNPID.nocontig.vcf.gz chr{1}_r20.7_annSNPID.vcf.gz" ::: {1..22} X

# Add contigs to make headers compatible
parallel -j 23 "bcftools reheader --fai GRCh38.fasta.fai -o chr{1}_r20.7_annSNPID.hdr.vcf.gz chr{1}_r20.7_annSNPID.nocontig.vcf.gz" ::: {1..22} X

# Gather vcfs
INPUTS=()
for CHR in {1..22} X; do INPUTS+=" -I chr${CHR}_r20.7_annSNPID.hdr.vcf.gz"; done

gatk GatherVcfs ${INPUTS[@]} -O ${NAMEBASE}_gathered.vcf.gz

# Turn into plink files
plink2 --vcf ${NAMEBASE}_gathered.vcf.gz --double-id --make-bed --out 00-${NAMEBASE}_annSNPID_r20.7


# The outcome is a set of imputed plink files.
